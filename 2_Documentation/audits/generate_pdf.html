def compute_laplacian_variance(frame: np.ndarray) -> float:
    """
    Computes image focus sharpness using the variance of Laplacian operator.
    
    Args:
        frame (np.ndarray): Input BGR frame matrix.
        
    Returns:
        float: Calculated variance. Higher values indicate sharper focus.
    """
    if frame is None or frame.size == 0:
        return 0.0
    
    if len(frame.shape) == 3 and frame.shape[2] == 3:
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    else:
        gray = frame
        
    laplacian = cv2.Laplacian(gray, cv2.CV_64F)
    return float(laplacian.var())

4. Defensive Programming Implementations

Defensive programming anticipates hardware inconsistencies, edge-case mathematical operations, and invalid inputs.

4.1 Inter-Shoulder Scale Normalization Guard

All 28 spatial features are normalized by inter-shoulder width. If examinee shoulders overlap or are occluded, raw distance becomes 0.0. The system enforces an epsilon guard (eps=1e-6) and fallback scaling constant to prevent ZeroDivisionError runtime crashes.

Compliant Code: Main/feature_extractor.py (Lines 31–43)
def compute_inter_shoulder_width(self, keypoints: np.ndarray) -> float:
    left_shoulder = (float(keypoints[5][0]), float(keypoints[5][1]))
    right_shoulder = (float(keypoints[6][0]), float(keypoints[6][1]))
    
    raw_width = euclidean_distance(left_shoulder, right_shoulder)
    
    # Defensive programming guard: enforce positive non-zero scale factor
    if raw_width <= self.eps or math.isnan(raw_width):
        logger.warning("Inter-shoulder width zero or NaN. Applying fallback constant (100.0 px).")
        return 100.0
    return raw_width

4.2 Floating-Point Overflow Guard in Platt Sigmoid Transformation

Platt scaling calculates \(P = \frac{1}{1 + e^{A \cdot f + B}}\). Large logit values can cause math.exp() overflow. The function includes numerical clamping for exponent values beyond \(\pm 80.0\).

Compliant Code: Main/classifier.py (Lines 26–36)
def platt_sigmoid_transform(raw_score: float, A: float, B: float) -> float:
    exponent = A * raw_score + B
    
    # Defensive overflow guard against float overflow in exp()
    if exponent > 80.0:
        return 0.0
    elif exponent < -80.0:
        return 1.0
        
    p_calibrated = 1.0 / (1.0 + math.exp(exponent))
    return float(np.clip(p_calibrated, 0.0, 1.0))

5. Error and Exception Handling

5.1 Custom System Exception Class Hierarchy

A custom exception hierarchy defined in Main/exceptions.py allows the system to distinguish between camera streaming failures, config errors, and deep learning inference crashes.

Compliant Code: Main/exceptions.py (Lines 11–26)
class ProctoringSystemError(Exception):
    """Base exception class for all proctoring system errors."""
    def __init__(self, message: str, error_code: Optional[int] = None) -> None:
        super().__init__(message)
        self.message = message
        self.error_code = error_code or 500

class CameraCaptureError(ProctoringSystemError):
    """Raised when video capture stream fails or disconnects."""
    def __init__(self, camera_id: str, details: str) -> None:
        message = f"Camera '{camera_id}' capture failed: {details}"
        super().__init__(message, error_code=501)
